#include <stdlib.h>

void swap(int* a, int* b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int findKthLargest(int* nums, int numsSize, int k) {
   
    srand(42);
    
    int target_idx = numsSize - k;
    int left = 0;
    int right = numsSize - 1;
    
    while (left <= right) {
        if (left == right) {
            return nums[left];
        }
        
       
        int pivot_idx = left + rand() % (right - left + 1);
        swap(&nums[pivot_idx], &nums[left]);
        int pivot = nums[left];
        
       
        int lt = left;       
        int gt = right;     
        int i = left + 1;    
        
        while (i <= gt) {
            if (nums[i] < pivot) {
                swap(&nums[i], &nums[lt]);
                lt++;
                i++;
            } else if (nums[i] > pivot) {
                swap(&nums[i], &nums[gt]);
                gt--;
               
            } else {
                i++;
            }
        }
        
        
        
        if (target_idx >= lt && target_idx <= gt) {
         
            return nums[target_idx];
        } else if (target_idx < lt) {
           
            right = lt - 1;
        } else {
       
            left = gt + 1;
        }
    }
    
    return -1; 
}
